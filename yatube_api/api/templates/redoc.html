""" Проект «Изгиб Питона» — классическая игра «Змейка» на Python с использованием принципов объектно-ориентированного программирования. """ import pygame import random from abc import ABC, abstractmethod from typing import List, Tuple, Optional # Constants for game configuration SCREEN_WIDTH = 640 SCREEN_HEIGHT = 480 GRID_SIZE = 20 GRID_WIDTH = SCREEN_WIDTH // GRID_SIZE GRID_HEIGHT = SCREEN_HEIGHT // GRID_SIZE BOARD_BACKGROUND_COLOR = (0, 0, 0) SNAKE_COLOR = (0, 255, 0) APPLE_COLOR = (255, 0, 0) # Direction vectors: (dx, dy) in grid units UP = (0, -1) DOWN = (0, 1) LEFT = (-1, 0) RIGHT = (1, 0) class GameObject(ABC): """ Базовый абстрактный класс для всех игровых объектов. Атрибуты: position: Позиция объекта на игровом поле (кортеж (x, y)). body_color: Цвет объекта в формате RGB. """ def __init__( self, position: Optional[Tuple[int, int]] = None, body_color: Optional[Tuple[int, int, int]] = None ) -> None: """ Инициализирует базовый игровой объект. Args: position: Начальная позиция объекта. По умолчанию — центр экрана. body_color: Цвет объекта в формате RGB. """ if position is None: position = (SCREEN_WIDTH // 2, SCREEN_HEIGHT // 2) self.position = position self.body_color = body_color @abstractmethod def draw(self, surface: pygame.Surface) -> None: """ Абстрактный метод для отрисовки объекта. Должен быть переопределён в дочерних классах. Args: surface: Поверхность Pygame для отрисовки. """ pass class Apple(GameObject): """ Класс, представляющий яблоко в игре. Атрибуты: body_color: Цвет яблока (красный по умолчанию). position: Текущая позиция яблока на игровом поле. """ def __init__(self) -> None: """ Инициализирует яблоко с красным цветом и случайной позицией. """ super().__init__(body_color=APPLE_COLOR) self.randomize_position() def randomize_position(self) -> None: """ Устанавливает случайную позицию яблока на игровом поле. Координаты кратны GRID_SIZE и находятся в пределах поля. """ x = random.randint(0, GRID_WIDTH - 1) * GRID_SIZE y = random.randint(0, GRID_HEIGHT - 1) * GRID_SIZE self.position = (x, y) def draw(self, surface: pygame.Surface) -> None: """ Отрисовывает яблоко на игровой поверхности. Args: surface: Поверхность Pygame для отрисовки. """ pygame.draw.rect( surface, self.body_color, (*self.position, GRID_SIZE, GRID_SIZE) ) class Snake(GameObject): """ Класс, представляющий змейку в игре. Атрибуты: length: Текущая длина змейки в сегментах. positions: Список кортежей с позициями всех сегментов тела. direction: Текущее направление движения (вектор). next_direction: Запланированное направление для следующего шага. body_color: Цвет змейки в формате RGB. last: Позиция последнего сегмента для стирания следа. """ def __init__(self) -> None: """ Инициализирует змейку в начальном состоянии. """ super().__init__(body_color=SNAKE_COLOR) self.length = 1 self.positions = [self.position] self.direction = RIGHT self.next_direction = None self.last = None def get_head_position(self) -> Tuple[int, int]: """ Возвращает позицию головы змейки. Returns: Кортеж с координатами (x, y) головы змейки. """ return self.positions[0] def update_direction(self) -> None: """ Обновляет направление движения змейки. Предотвращает разворот на 180 градусов. Применяет next_direction, если он задан и не противоположен текущему. """ if self.next_direction: # Prevent 180-degree turns if (self.next_direction[0] * -1, self.next_direction[1] * -1) != self.direction: self.direction = self.next_direction self.next_direction = None def move(self) -> None: """ Обновляет позицию змейки на одну ячейку в текущем направлении. Обрабатывает: - Телепортацию через края экрана (обёртывание) - Проверку столкновения с самим собой - Добавление новой головы и удаление хвоста - Сохранение позиции хвоста для стирания следа """ head_x, head_y = self.get_head_position() dx, dy = self.direction # Calculate new head position with screen wrapping new_x = (head_x + dx * GRID_SIZE) % SCREEN_WIDTH new_y = (head_y + dy * GRID_SIZE) % SCREEN_HEIGHT new_head = (new_x, new_y) # Check for self-collision (exclude head and neck) if new_head in self.positions[2:]: self.reset() return # Store tail position for erasing trail before modifying list if len(self.positions) > self.length: self.last = self.positions[-1] else: self.last = None # Add new head to the beginning of positions list self.positions.insert(0, new_head) # Remove tail if snake hasn't grown (didn't eat apple) if len(self.positions) > self.length: self.positions.pop() def draw(self, surface: pygame.Surface) -> None: """ Отрисовывает змейку и стирает след от предыдущей позиции хвоста. Args: surface: Поверхность Pygame для отрисовки. """ # Draw all body segments for position in self.positions: pygame.draw.rect( surface, self.body_color, (*position, GRID_SIZE, GRID_SIZE) ) # Erase the trail of the old tail position if self.last: pygame.draw.rect( surface, BOARD_BACKGROUND_COLOR, (*self.last, GRID_SIZE, GRID_SIZE) ) def reset(self) -> None: """ Сбрасывает змейку в начальное состояние. Восстанавливает: - Длину до 1 сегмента - Позицию в центр экрана - Направление движения (случайное) """ self.length = 1 self.positions = [(SCREEN_WIDTH // 2, SCREEN_HEIGHT // 2)] self.direction = random.choice([UP, DOWN, LEFT, RIGHT]) self.next_direction = None self.last = None def handle_keys(snake: Snake) -> None: """ Обрабатывает события ввода от пользователя. Изменяет next_direction змейки при нажатии стрелок. Завершает игру при закрытии окна. Args: snake: Экземпляр змейки для управления. """ for event in pygame.event.get(): if event.type == pygame.QUIT: pygame.quit() exit() elif event.type == pygame.KEYDOWN: if event.key == pygame.K_UP: snake.next_direction = UP elif event.key == pygame.K_DOWN: snake.next_direction = DOWN elif event.key == pygame.K_LEFT: snake.next_direction = LEFT elif event.key == pygame.K_RIGHT: snake.next_direction = RIGHT def main() -> None: """ Основная функция игры. Запускает и поддерживает игровой цикл. Инициализирует Pygame, создаёт игровые объекты и обрабатывает: - Ввод пользователя - Обновление состояния игры - Проверку коллизий - Отрисовку объектов - Управление частотой кадров """ pygame.init() # Create game window screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT)) pygame.display.set_caption('Изгиб Питона') # Create clock for controlling frame rate clock = pygame.time.Clock() # Create game objects snake = Snake() apple = Apple() # Main game loop while True: # Handle user input handle_keys(snake) # Update snake direction and move snake.update_direction() snake.move() # Check if snake ate the apple if snake.get_head_position() == apple.position: snake.length += 1 # Respawn apple at random position (not on snake) apple.randomize_position() while apple.position in snake.positions: apple.randomize_position() # Clear screen with background color screen.fill(BOARD_BACKGROUND_COLOR) # Draw game objects apple.draw(screen) snake.draw(screen) # Update display pygame.display.update() # Control game speed (20 FPS) clock.tick(20) if __name__ == '__main__': main()
